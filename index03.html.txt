response:

Here’s a complete one‑page webapp implementation of **Simple Tic‑Tac‑Toe** with the settings overlay, one‑player vs computer mode, first‑move toggle, and end‑game result overlay. You can copy this into a single `.html` file and open it in your browser:

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Simple Tic-Tac-Toe</title>
  <style>
    body { font-family: Arial, sans-serif; margin:0; display:flex; justify-content:center; align-items:center; height:100vh; background:#f0f0f0; }
    #game { display:grid; grid-template-columns:repeat(3,100px); grid-template-rows:repeat(3,100px); gap:5px; }
    .cell { background:#fff; display:flex; justify-content:center; align-items:center; font-size:2em; cursor:pointer; }
    .overlay { position:fixed; top:0; left:0; width:100%; height:100%; background:rgba(0,0,0,0.7); display:flex; justify-content:center; align-items:center; }
    .overlay-content { background:#fff; padding:20px; border-radius:8px; text-align:center; }
    button { margin:5px; padding:10px 20px; }
  </style>
</head>
<body>
  <div id="game"></div>

  <!-- Settings Overlay -->
  <div id="settings" class="overlay">
    <div class="overlay-content">
      <h2>Simple Tic-Tac-Toe</h2>
      <p>Select Mode:</p>
      <button onclick="startGame('two')">Two Players</button>
      <button onclick="showOnePlayerOptions()">One Player</button>
      <div id="onePlayerOptions" style="display:none;">
        <p>Who moves first?</p>
        <button onclick="startGame('one','player')">Player First</button>
        <button onclick="startGame('one','computer')">Computer First</button>
      </div>
    </div>
  </div>

  <!-- Result Overlay -->
  <div id="result" class="overlay" style="display:none;">
    <div class="overlay-content">
      <h2 id="resultText"></h2>
      <button onclick="resetGame()">Play Again</button>
    </div>
  </div>

<script>
  const gameBoard = document.getElementById('game');
  let board, currentPlayer, mode, firstMove;

  function initBoard() {
    board = Array(9).fill('');
    gameBoard.innerHTML = '';
    for (let i=0;i<9;i++) {
      const cell = document.createElement('div');
      cell.className = 'cell';
      cell.dataset.index = i;
      cell.addEventListener('click', handleMove);
      gameBoard.appendChild(cell);
    }
  }

  function showOnePlayerOptions() {
    document.getElementById('onePlayerOptions').style.display = 'block';
  }

  function startGame(selectedMode, first) {
    mode = selectedMode;
    firstMove = first;
    document.getElementById('settings').style.display = 'none';
    initBoard();
    currentPlayer = 'X';
    if (mode==='one' && firstMove==='computer') {
      computerMove();
    }
  }

  function handleMove(e) {
    const idx = e.target.dataset.index;
    if (board[idx]!=='' || (mode==='one' && currentPlayer==='O')) return;
    board[idx] = currentPlayer;
    e.target.textContent = currentPlayer;
    if (checkWinner()) return endGame(currentPlayer+' wins!');
    if (board.every(cell=>cell!=='')) return endGame('It\'s a draw!');
    currentPlayer = currentPlayer==='X'?'O':'X';
    if (mode==='one' && currentPlayer==='O') {
      setTimeout(computerMove,500);
    }
  }

  function computerMove() {
    let emptyIndices = board.map((v,i)=>v===''?i:null).filter(v=>v!==null);
    let idx = emptyIndices[Math.floor(Math.random()*emptyIndices.length)];
    board[idx] = 'O';
    gameBoard.children[idx].textContent = 'O';
    if (checkWinner()) return endGame('Computer wins!');
    if (board.every(cell=>cell!=='')) return endGame('It\'s a draw!');
    currentPlayer = 'X';
  }

  function checkWinner() {
    const wins = [
      [0,1,2],[3,4,5],[6,7,8],
      [0,3,6],[1,4,7],[2,5,8],
      [0,4,8],[2,4,6]
    ];
    return wins.some(combo=>{
      const [a,b,c] = combo;
      return board[a] && board[a]===board[b] && board[a]===board[c];
    });
  }

  function endGame(message) {
    document.getElementById('resultText').textContent = message;
    document.getElementById('result').style.display = 'flex';
  }

  function resetGame() {
    document.getElementById('result').style.display = 'none';
    document.getElementById('settings').style.display = 'flex';
  }
</script>
</body>
</html>
```

This webapp flow works like this:
- At launch, the **settings overlay** appears with options for one player or two players.
- In one player mode, you can choose whether the player moves first or the computer moves first.
- After each game, the **result overlay** announces the outcome (win, loss, or draw) and then returns to the settings screen.

        (function() {
            'use strict';

            const settingsOverlay = document.getElementById('settingsOverlay');

            const gameContainer = document.getElementById('gameContainer');
            const boardEl = document.getElementById('board');
            const turnIndicator = document.getElementById('turnIndicator');
            const turnSymbol = document.getElementById('turnSymbol');
            const btnStart = document.getElementById('btnStart');
            const btnSettings = document.getElementById('btnSettings');
            const btnPlayAgain = document.getElementById('btnPlayAgain');
            const modeToggle = document.getElementById('modeToggle');
            const firstToggle = document.getElementById('firstToggle');
            const firstMoveGroup = document.getElementById('firstMoveGroup');
            const firstLabelOff = document.getElementById('firstLabelOff');
            const firstLabelOn = document.getElementById('firstLabelOn');
            const resultTitle = document.getElementById('resultTitle');
            const resultSub = document.getElementById('resultSub');
            const resultSymbol = document.getElementById('resultSymbol');

            // ─── state ───
            let mode = 'onePlayer'; // 'onePlayer' | 'twoPlayers'
            let firstMove = 'player'; // 'player' | 'computer'  (only for onePlayer)
            let board = Array(9).fill(null);
            let currentPlayer = 'X'; // 'X' always goes first
            let gameActive = false;
            let gameOver = false;
            let isComputerTurn = false;
            let winCells = [];

            // ─── constants ───
            const HUMAN = 'X';
            const COMPUTER = 'O';

            // ─── render board ───
            function renderBoard() {
                boardEl.innerHTML = '';
                for (let i = 0; i < 9; i++) {
                    const cell = document.createElement('div');
                    cell.className = 'cell';
                    cell.dataset.index = i;
                    const value = board[i];
                    if (value === 'X') {
                        cell.innerHTML = '<span class="x-mark">✕</span>';
                        cell.classList.add('taken');
                    } else if (value === 'O') {
                        cell.innerHTML = '<span class="o-mark">◯</span>';
                        cell.classList.add('taken');
                    }
                    if (gameOver) cell.classList.add('game-over');
                    if (winCells.includes(i)) cell.classList.add('win-cell');
                    cell.addEventListener('click', () => handleCellClick(i));
                    boardEl.appendChild(cell);
                }
                updateTurnIndicator();
            }

            function updateTurnIndicator() {
                if (gameOver) {
                    turnIndicator.innerHTML = `<span>Game Over</span>`;
                    return;
                }
                const symbol = currentPlayer === 'X' ? 'X' : 'O';
                const cls = symbol === 'X' ? 'x' : 'o';
                turnIndicator.innerHTML = `
                    <span>Turn</span>
                    <span class="turn-symbol ${cls}">${symbol}</span>
                `;
                turnSymbol.className = `turn-symbol ${cls}`;
                turnSymbol.textContent = symbol;
            }

            // ─── win detection ───
            function checkWinner() {
                const lines = [
                    [0, 1, 2],
                    [3, 4, 5],
                    [6, 7, 8],
                    [0, 3, 6],
                    [1, 4, 7],
                    [2, 5, 8],
                    [0, 4, 8],
                    [2, 4, 6]
                ];
                for (const line of lines) {
                    const [a, b, c] = line;
                    if (board[a] && board[a] === board[b] && board[a] === board[c]) {
                        return { winner: board[a], cells: line };
                    }
                }
                if (board.every(v => v !== null)) {
                    return { winner: 'draw', cells: [] };
                }
                return null;
            }

            function isBoardFull() {
                return board.every(v => v !== null);
            }

            function getEmptyCells() {
                return board.map((v, i) => v === null ? i : null).filter(v => v !== null);
            }

            // ─── minimax AI ───
            function minimax(b, depth, isMaximizing) {
                const result = checkWinnerBoard(b);
                if (result === 'X') return 10 - depth;
                if (result === 'O') return depth - 10;
                if (result === 'draw') return 0;

                if (isMaximizing) {
                    let best = -Infinity;
                    for (let i = 0; i < 9; i++) {
                        if (b[i] === null) {
                            b[i] = 'X';
                            best = Math.max(best, minimax(b, depth + 1, false));
                            b[i] = null;
                        }
                    }
                    return best;
                } else {
                    let best = Infinity;
                    for (let i = 0; i < 9; i++) {
                        if (b[i] === null) {
                            b[i] = 'O';
                            best = Math.min(best, minimax(b, depth + 1, true));
                            b[i] = null;
                        }
                    }
                    return best;
                }
            }

            function checkWinnerBoard(b) {
                const lines = [
                    [0, 1, 2],
                    [3, 4, 5],
                    [6, 7, 8],
                    [0, 3, 6],
                    [1, 4, 7],
                    [2, 5, 8],
                    [0, 4, 8],
                    [2, 4, 6]
                ];
                for (const line of lines) {
                    const [a, bC, c] = line;
                    if (b[a] && b[a] === b[bC] && b[a] === b[c]) {
                        return b[a];
                    }
                }
                if (b.every(v => v !== null)) return 'draw';
                return null;
            }

            function getComputerMove() {
                const empty = getEmptyCells();
                if (empty.length === 0) return null;

                // If computer is O (minimizer)
                let bestScore = Infinity;
                let bestMove = empty[0];

                for (const i of empty) {
                    board[i] = COMPUTER;
                    const score = minimax(board, 0, true);
                    board[i] = null;
                    if (score < bestScore) {
                        bestScore = score;
                        bestMove = i;
                    }
                }
                return bestMove;
            }

            // ─── game logic ───
            function handleCellClick(index) {
                if (!gameActive || gameOver) return;
                if (isComputerTurn) return;
                if (board[index] !== null) return;

                // Human move
                makeMove(index, HUMAN);

                // Check win after human move
                if (gameOver) return;

                // If one player mode and game still active, trigger computer turn
                if (mode === 'onePlayer' && gameActive && !gameOver) {
                    isComputerTurn = true;
                    renderBoard();
                    // delay computer move for a better feel
                    setTimeout(() => {
                        if (gameActive && !gameOver) {
                            computerMove();
                        }
                        isComputerTurn = false;
                    }, 280);
                }
            }

            function makeMove(index, player) {
                if (board[index] !== null) return false;
                board[index] = player;
                currentPlayer = (player === 'X' ? 'O' : 'X');
                renderBoard();

                const result = checkWinner();
                if (result) {
                    gameOver = true;
                    gameActive = false;
                    if (result.winner !== 'draw') {
                        winCells = result.cells;
                        renderBoard();
                    }
                    showResult(result.winner);
                    return true;
                }
                return true;
            }

            function computerMove() {
                if (!gameActive || gameOver) return;
                if (isBoardFull()) return;

                const move = getComputerMove();
                if (move === null) return;

                board[move] = COMPUTER;
                currentPlayer = HUMAN;
                renderBoard();

                const result = checkWinner();
                if (result) {
                    gameOver = true;
                    gameActive = false;
                    if (result.winner !== 'draw') {
                        winCells = result.cells;
                        renderBoard();
                    }
                    showResult(result.winner);
                }
            }

            function showResult(winner) {
                const isOnePlayer = mode === 'onePlayer';
                let title, sub, symbol;

                if (winner === 'draw') {
                    title = "It's a Draw!";
                    sub = "Well played!";
                    symbol = '🤝';
                    resultTitle.className = 'result-title draw';
                } else if (winner === 'X') {
                    if (isOnePlayer) {
                        title = '🎉 You Win!';
                        sub = 'Brilliant! You beat the computer!';
                        symbol = '🏆';
                        resultTitle.className = 'result-title win';
                    } else {
                        title = '🎉 Player 1 Wins!';
                        sub = 'Player 1 (✕) takes the victory!';
                        symbol = '🏆';
                        resultTitle.className = 'result-title win';
                    }
                } else { // winner === 'O'
                    if (isOnePlayer) {
                        title = '💻 Computer Wins!';
                        sub = 'Better luck next time!';
                        symbol = '🤖';
                        resultTitle.className = 'result-title lose';
                    } else {
                        title = '🎉 Player 2 Wins!';
                        sub = 'Player 2 (◯) takes the victory!';
                        symbol = '🏆';
                        resultTitle.className = 'result-title win';
                    }
                }

                resultTitle.textContent = title;
                resultSub.textContent = sub;
                resultSymbol.textContent = symbol;

                // show result overlay after a short delay
                setTimeout(() => {
                    resultOverlay.classList.add('active');
                }, 400);
            }

            // ─── start game ───
            function startGame() {
                // reset state
                board = Array(9).fill(null);
                currentPlayer = 'X';
                gameActive = true;
                gameOver = false;
                winCells = [];
                isComputerTurn = false;

                // hide overlays
                settingsOverlay.classList.remove('active');
                resultOverlay.classList.remove('active');
                gameContainer.classList.add('active');

                renderBoard();

                // Determine first move
                if (mode === 'onePlayer') {
                    if (firstMove === 'computer') {
                        // Computer (O) goes first
                        isComputerTurn = true;
                        setTimeout(() => {
                            if (gameActive && !gameOver) {
                                computerMove();
                            }
                            isComputerTurn = false;
                        }, 350);
                    } else {
                        // Player (X) goes first
                        isComputerTurn = false;
                    }
                } else {
                    // Two players: X goes first
                    isComputerTurn = false;
                }
                updateTurnIndicator();
            }

            // ─── reset to settings ───
            function goToSettings() {
                gameContainer.classList.remove('active');
                resultOverlay.classList.remove('active');
                settingsOverlay.classList.add('active');
                gameActive = false;
                gameOver = false;
                isComputerTurn = false;
                board = Array(9).fill(null);
                winCells = [];
                renderBoard();
            }

            // ─── UI event handlers ───

            // Mode toggle
            modeToggle.addEventListener('click', (e) => {
                const btn = e.target.closest('button');
                if (!btn) return;
                const newMode = btn.dataset.mode;
                if (newMode === mode) return;
                mode = newMode;
                modeToggle.querySelectorAll('button').forEach(b => b.classList.remove('active'));
                btn.classList.add('active');
                // show/hide first move toggle
                if (mode === 'onePlayer') {
                    firstMoveGroup.style.display = 'block';
                } else {
                    firstMoveGroup.style.display = 'none';
                }
            });

            // First move toggle
            firstToggle.addEventListener('click', () => {
                const isActive = firstToggle.classList.contains('active');
                if (isActive) {
                    firstToggle.classList.remove('active');
                    firstMove = 'computer';
                    firstLabelOff.style.color = '#ffd200';
                    firstLabelOn.style.color = 'rgba(255,255,255,0.35)';
                } else {
                    firstToggle.classList.add('active');
                    firstMove = 'player';
                    firstLabelOff.style.color = 'rgba(255,255,255,0.35)';
                    firstLabelOn.style.color = '#ffd200';
                }
            });

            // Start button
            btnStart.addEventListener('click', startGame);

            // Settings button (during game)
            btnSettings.addEventListener('click', goToSettings);

            // Play again button (after result)
            btnPlayAgain.addEventListener('click', goToSettings);

            // ─── init ───
            function init() {
                // Set default mode
                modeToggle.querySelector('[data-mode="onePlayer"]').classList.add('active');
                modeToggle.querySelector('[data-mode="twoPlayers"]').classList.remove('active');
                mode = 'onePlayer';
                firstMoveGroup.style.display = 'block';
                firstToggle.classList.add('active');
                firstMove = 'player';
                firstLabelOff.style.color = 'rgba(255,255,255,0.35)';
                firstLabelOn.style.color = '#ffd200';

                board = Array(9).fill(null);
                renderBoard();
                settingsOverlay.classList.add('active');
                gameContainer.classList.remove('active');
                resultOverlay.classList.remove('active');
                gameActive = false;
                gameOver = false;
                winCells = [];
            }

            init();

        })();
    </script>

</body>

### Gameplay Overview

Here's how the game works and what you can expect:

- **Setup & Modes**: The game starts with a settings overlay. You can choose **1 Player** (vs computer) or **2 Players** (local multiplayer). For one-player mode, toggle who moves first — you or the computer.
- **Game Flow**: Click a cell to place your mark (X). The computer (O) will respond automatically in one-player mode. In two-player mode, players take turns. The turn indicator at the top shows whose turn it is.
- **Result & Reset**: After a win or draw, a result overlay appears with a clear message. Click **"Play Again"** to return to the settings screen, where you can change mode or first-move preference before starting a new game.

The computer uses a minimax algorithm to make optimal moves, providing a challenging experience in one-player mode.

---

**Optimization Tip:** You can adjust the computer's difficulty by modifying the `minimax` function's depth scoring (lines ~240-260) or adding randomness to move selection.
